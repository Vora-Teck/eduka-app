        // Tools Data

        // Dictionary (using Free Dictionary API)
        async function searchDictionary() {
            const word = $('#dict-search').val().trim().toLowerCase();
            if (!word) return;
            
            const resultDiv = $('#dictionary-result');
            resultDiv.html('<div class="animate-pulse">Looking up...</div>');
            
            try {
                const res = await fetch(`https://api.dictionaryapi.dev/api/v2/entries/en/${word}`);
                const data = await res.json();
                
                if (Array.isArray(data)) {
                    const entry = data[0];
                    let html = `
                        <div class="text-left">
                            <div class="text-4xl font-bold mb-2">${entry.word}</div>
                            <div class="text-emerald-600">${entry.phonetic || ''}</div>
                            <div class="mt-6 space-y-6">`;
                    
                    entry.meanings.forEach(meaning => {
                        html += `<div>
                            <div class="font-medium text-violet-600">${meaning.partOfSpeech}</div>`;
                        meaning.definitions.forEach((def, i) => {
                            html += `<div class="mt-3 pl-4 border-l-2 border-gray-200 dark:border-gray-700">
                                <div>${def.definition}</div>
                                \( {def.example ? `<div class="text-xs text-gray-500 mt-2">" \){def.example}"</div>` : ''}
                            </div>`;
                        });
                        html += `</div>`;
                    });
                    html += `</div></div>`;
                    resultDiv.html(html);
                } else {
                    resultDiv.html(`<div class="text-red-500">Word not found</div>`);
                }
            } catch(e) {
                resultDiv.html(`<div class="text-red-500">Connection error. Try again.</div>`);
            }
        }

        // Scientific Calculator
        let calcDisplay = '0';
        let prevValue = '';
        let operator = '';
        
        function initCalculator() {
            const buttons = [
                'C','+/-','%','÷',
                '7','8','9','×',
                '4','5','6','-',
                '1','2','3','+',
                '0','.','sin','cos',
                'tan','log','√','='
            ];
            
            let html = '';
            buttons.forEach(btn => {
                let extraClass = '';
                if (['÷','×','-','+','='].includes(btn)) extraClass = 'bg-emerald-600 text-white';
                if (['C','+/-','%'].includes(btn)) extraClass = 'text-gray-400';
                
                html += `<button onclick="handleCalcInput('${btn}')" 
                                class="h-14 rounded-2xl text-xl font-medium \( {extraClass}"> \){btn}</button>`;
            });
            $('#calculator-modal .grid').html(html);
            calcDisplay = '0';
            updateCalcDisplay();
        }

        function handleCalcInput(val) {
            if (val === 'C') {
                calcDisplay = '0';
                prevValue = '';
                operator = '';
            } else if (['+','-','×','÷'].includes(val)) {
                prevValue = calcDisplay;
                operator = val;
                calcDisplay = '0';
            } else if (val === '=') {
                calculateResult();
            } else if (['sin','cos','tan','log','√'].includes(val)) {
                calculateScientific(val);
            } else {
                if (calcDisplay === '0' && val !== '.') calcDisplay = val;
                else calcDisplay += val;
            }
            updateCalcDisplay();
        }

        function calculateResult() {
            let result = 0;
            const a = parseFloat(prevValue);
            const b = parseFloat(calcDisplay);
            
            switch(operator) {
                case '+': result = a + b; break;
                case '-': result = a - b; break;
                case '×': result = a * b; break;
                case '÷': result = a / b; break;
            }
            calcDisplay = result.toString();
        }

        function calculateScientific(func) {
            const num = parseFloat(calcDisplay);
            switch(func) {
                case 'sin': calcDisplay = Math.sin(num * Math.PI/180).toFixed(6); break;
                case 'cos': calcDisplay = Math.cos(num * Math.PI/180).toFixed(6); break;
                case 'tan': calcDisplay = Math.tan(num * Math.PI/180).toFixed(6); break;
                case 'log': calcDisplay = Math.log10(num).toFixed(6); break;
                case '√': calcDisplay = Math.sqrt(num).toFixed(6); break;
            }
        }

        function updateCalcDisplay() {
            $('#calc-display').text(calcDisplay);
        }

        // Unit Converter
        const converters = {
            length: { units: ["Meter", "Kilometer", "Centimeter", "Inch", "Foot"], factor: [1, 0.001, 100, 39.37, 3.28] },
            weight: { units: ["Kilogram", "Gram", "Pound", "Ounce"], factor: [1, 1000, 2.205, 35.27] },
            temperature: { units: ["Celsius", "Fahrenheit", "Kelvin"], factor: null },
            volume: { units: ["Liter", "Milliliter", "Gallon"], factor: [1, 1000, 0.264] }
        };

        function loadConverterUnits() {
            const category = $('#converter-category').val();
            const data = converters[category];
            
            let fromHTML = '', toHTML = '';
            data.units.forEach((unit, i) => {
                fromHTML += `<option value="\( {i}"> \){unit}</option>`;
                toHTML += `<option value="\( {i}"> \){unit}</option>`;
            });
            
            $('#from-unit').html(fromHTML);
            $('#to-unit').html(toHTML);
            convertUnits();
        }

        function convertUnits() {
            const category = $('#converter-category').val();
            const val = parseFloat($('#input-value').val()) || 0;
            const from = parseInt($('#from-unit').val());
            const to = parseInt($('#to-unit').val());
            
            let result = val;
            
            if (category === 'temperature') {
                // Simple C to F conversion for demo
                if (from === 0 && to === 1) result = (val * 9/5) + 32;
                else if (from === 1 && to === 0) result = (val - 32) * 5/9;
            } else {
                const data = converters[category];
                result = val * (data.factor[to] / data.factor[from]);
            }
            
            $('#output-value').val(result.toFixed(4));
        }

        // Math Tables
        function showTable(type) {
            $('.math-tab-btn').removeClass('active');
            \( (`.math-tab-btn:contains( \){type.charAt(0).toUpperCase() + type.slice(1)})`).addClass('active');
            
            let html = `<table class="w-full text-sm"><thead><tr class="border-b"><th class="py-3 text-left">x</th><th class="py-3 text-right">Value</th></tr></thead><tbody>`;
            
            for (let i = 0; i <= 360; i += 15) {
                let val = 0;
                if (type === 'sine') val = Math.sin(i * Math.PI/180).toFixed(4);
                else if (type === 'cosine') val = Math.cos(i * Math.PI/180).toFixed(4);
                else if (type === 'tangent') val = Math.tan(i * Math.PI/180).toFixed(4);
                else if (type === 'log') val = Math.log10(i || 1).toFixed(4);
                
                html += `<tr class="border-b last:border-none"><td class="py-3">\( {i}</td><td class="py-3 text-right font-medium"> \){val}</td></tr>`;
            }
            html += `</tbody></table>`;
            $('#math-table-content').html(html);
        }

        // Graph Plotter (using Canvas)
        let canvas, ctx;
        function plotGraph() {
            canvas = document.getElementById('graph-canvas');
            ctx = canvas.getContext('2d');
            canvas.width = canvas.offsetWidth;
            canvas.height = canvas.offsetHeight;
            
            ctx.clearRect(0, 0, canvas.width, canvas.height);
            
            // Draw axes
            ctx.strokeStyle = '#10b981';
            ctx.lineWidth = 2;
            ctx.beginPath();
            ctx.moveTo(40, 40);
            ctx.lineTo(40, canvas.height - 40);
            ctx.lineTo(canvas.width - 40, canvas.height - 40);
            ctx.stroke();
            
            // Plot sample function (x^2)
            ctx.strokeStyle = '#eab308';
            ctx.lineWidth = 3;
            ctx.beginPath();
            
            for (let x = -10; x <= 10; x += 0.1) {
                const y = eval($('#graph-function').val().replace(/x/g, x)) || 0;
                const px = 40 + (x + 10) * (canvas.width - 80) / 20;
                const py = canvas.height - 40 - (y + 10) * (canvas.height - 80) / 20;
                if (x === -10) ctx.moveTo(px, py);
                else ctx.lineTo(px, py);
            }
            ctx.stroke();
        }

        function showElementInfo(num) {
